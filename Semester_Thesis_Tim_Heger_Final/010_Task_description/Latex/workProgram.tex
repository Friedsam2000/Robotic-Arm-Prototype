\subsection{Work program (8 weeks)}
\begin{itemize}
	\item (1.5 weeks) Literature review and study of model order reduction by Krylov-subspace methods for MIMO models \parencite{Wilson_1974,Beattie_2014_Survey,Druskin_2014,Beattie_2009_TrustRegion,Gallivan_2004_MIMO,VanDooren_2008,VanDooren_2010}
	\item (1.5 weeks) Implementation of the reduction methods from the previous point into the sssMOR toolbox.
	\item (0.5 weeks) Application of the reduction methods for a series of MIMO benchmark models.		
	\item (1.5 week) Analysis of the resulting tangential directions and investigations on their physical interpretation.
	\item (3 weeks) Thesis Report
\end{itemize}

\subsection{Core questions}
\begin{itemize}
	\item Is there any pattern in the choice of tangential directions for $\Htwo$ and residual approaches?
	\item How can we interpret $G(s)r_t$, $\trans{l_t}G(s)$ and $\trans{l_t}G(s)r_t$?
\end{itemize}
